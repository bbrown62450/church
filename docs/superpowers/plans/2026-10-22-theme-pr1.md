// Test helpers for Theme PR 1: the committed color tokens as globals.css holds them, and WCAG contrast.
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { fileURLToPath } from "node:url";

/** The frontend directory. */
export const FRONTEND = fileURLToPath(new URL("../../", import.meta.url));
export const read = (path: string) => readFileSync(join(FRONTEND, path), "utf8");

export type Tokens = Record<string, string>;

const tokensOf = (section: string): Tokens =>
  Object.fromEntries([...section.matchAll(/--([\w-]+): (#[0-9a-f]{6});/g)].map((m) => [m[1], m[2]]));

/** The generated block of globals.css, from `/* theme:start` to `/* theme:end *\/`. */
export function themeBlock(): string {
  const css = read("src/app/globals.css");
  const end = "/* theme:end */";
  return css.slice(css.indexOf("/* theme:start"), css.indexOf(end) + end.length);
}

/** The committed tokens: light (:root), dark (under prefers-color-scheme: dark) and paper (.paper). */
export function committedTokens(): Record<"light" | "dark" | "paper", Tokens> {
  const block = themeBlock();
  const darkAt = block.indexOf("@media (prefers-color-scheme: dark)");
  const paperAt = block.indexOf(".paper {");
  return {
    light: tokensOf(block.slice(0, darkAt)),
    dark: tokensOf(block.slice(darkAt, paperAt)),
    paper: tokensOf(block.slice(paperAt)),
  };
}

function rgb(hex: string): number[] {
  const n = parseInt(hex.slice(1), 16);
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
}

/** `top` at `alpha` over `bottom`, as the browser paints a Tailwind `/10` color. */
export function over(top: string, alpha: number, bottom: string): string {
  const [a, b] = [rgb(top), rgb(bottom)];
  return "#" + a.map((c, i) => Math.round(c * alpha + b[i] * (1 - alpha)).toString(16).padStart(2, "0")).join("");
}

function luminance(hex: string): number {
  const [r, g, b] = rgb(hex).map((c) => {
    const s = c / 255;
    return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  });
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}

/** The WCAG 2 contrast ratio of two `#rrggbb` colors. */
export function contrast(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
}

// Theme PR 1: the committed Material 3 tokens, their contrast in both modes, and the paper scope.
import { describe, expect, it } from "vitest";

import { committedTokens, contrast, over, read, themeBlock } from "@/test/theme";

import { SEED, themeCss, themeTokens } from "../../../scripts/generate-theme.mjs";

const MODES = committedTokens();
const SURFACES = ["background", "card", "popover", "muted", "surface-container"];
/** [text, background] pairs the components draw; each needs 4.5:1 (WCAG AA, normal text). */
const TEXT_PAIRS: [string, string][] = [
  ...SURFACES.flatMap((s) => [["foreground", s], ["muted-foreground", s], ["primary", s], ["destructive", s], ["warning", s]] as [string, string][]),
  ["accent-foreground", "accent"],
  ["primary-foreground", "primary"],
  ["secondary-foreground", "secondary"],
  ["primary-container-foreground", "primary-container"],
];
/** Field and outlined-button edges and focus rings: 3:1 (WCAG 1.4.11). */
const EDGE_PAIRS: [string, string][] = SURFACES.flatMap((s) => [["input", s], ["ring", s]] as [string, string][]);

const below = (mode: "light" | "dark", pairs: [string, string][], min: number) =>
  pairs
    .map(([fg, bg]) => ({ pair: `${fg} on ${bg}`, ratio: contrast(MODES[mode][fg], MODES[mode][bg]) }))
    .filter((r) => !(r.ratio >= min));

describe("the Material 3 tokens (Theme PR 1)", () => {
  it("are what scripts/generate-theme.mjs makes from the seed (run it with --write after a change)", () => {
    expect(SEED).toBe("#006A60");
    expect(themeBlock()).toBe(themeCss());
    expect(MODES.light).toEqual(themeTokens(false));
    expect(MODES.dark).toEqual(themeTokens(true));
  });

  it("hold every token in light, dark and paper, with paper equal to light", () => {
    expect(Object.keys(MODES.light).length).toBeGreaterThan(30);
    expect(Object.keys(MODES.dark)).toEqual(Object.keys(MODES.light));
    expect(MODES.paper).toEqual(MODES.light);
    expect(MODES.dark.background).not.toBe(MODES.light.background);
  });

  it.each(["light", "dark"] as const)("reach WCAG AA (4.5:1) for every text pair in %s", (mode) => {
    expect(below(mode, TEXT_PAIRS, 4.5)).toEqual([]);
  });

  it.each(["light", "dark"] as const)("reach 3:1 for field edges and focus rings in %s", (mode) => {
    expect(below(mode, EDGE_PAIRS, 3)).toEqual([]);
  });

  it.each([
    ["light", 0.1],
    ["dark", 0.2],

    const t = MODES[mode];

  });

// The app's Material 3 colors (Theme PR 1). Generates the color tokens in
// src/app/globals.css from one seed with Google's Material Color Utilities
// (the "tonal spot" scheme, standard contrast), so the page needs no color
// library at run time. Run `node scripts/generate-theme.mjs --write` after
// changing the seed or the role map; src/lib/theme.test.ts fails while the
// committed block differs from what this prints.
import { readFileSync, writeFileSync } from "node:fs";
import { fileURLToPath } from "node:url";

import {
  Blend,
  Hct,
  MaterialDynamicColors as M,
  SchemeTonalSpot,
  TonalPalette,
  argbFromHex,
  hexFromArgb,
} from "@material/material-color-utilities";

/** The owner's chosen preview (Option B, teal). */
export const SEED = "#006A60";
/** The warning text color's source, harmonized toward the seed (Material's "custom color" recipe). */
export const WARNING_SOURCE = "#B45309";

/** shadcn's variable → the Material 3 role it takes, in both modes. */
const ROLES = {
  background: "surface",
  foreground: "onSurface",
  card: "surfaceContainerLow",
  "card-foreground": "onSurface",
  popover: "surfaceContainer",
  "popover-foreground": "onSurface",
  primary: "primary",
  "primary-foreground": "onPrimary",
  "primary-container": "primaryContainer",
  "primary-container-foreground": "onPrimaryContainer",
  secondary: "secondaryContainer",
  "secondary-foreground": "onSecondaryContainer",
  muted: "surfaceContainerHigh",
  "muted-foreground": "onSurfaceVariant",
  accent: "surfaceContainerHighest",
  "accent-foreground": "onSurface",
  destructive: "error",
  border: "outlineVariant",
  input: "outline",
  ring: "primary",
  "surface-container": "surfaceContainer",
  "chart-1": "primary",
  "chart-2": "secondary",
  "chart-3": "tertiary",
  "chart-4": "primaryContainer",
  "chart-5": "tertiaryContainer",
  sidebar: "surfaceContainerLow",
  "sidebar-foreground": "onSurface",
  "sidebar-primary": "primary",
  "sidebar-primary-foreground": "onPrimary",
  "sidebar-accent": "secondaryContainer",
  "sidebar-accent-foreground": "onSecondaryContainer",
  "sidebar-border": "outlineVariant",
  "sidebar-ring": "primary",
};

/**
 * Every token for one mode, as `#rrggbb`.
 * @param {boolean} dark
 * @returns {Record<string, string>}
 */
export function themeTokens(dark) {
  const scheme = new SchemeTonalSpot(Hct.fromInt(argbFromHex(SEED)), dark, 0);
  /** @type {Record<string, string>} */
  const tokens = {};
  for (const [name, role] of Object.entries(ROLES)) tokens[name] = hexFromArgb(M[role].getArgb(scheme));
  const warning = TonalPalette.fromInt(Blend.harmonize(argbFromHex(WARNING_SOURCE), argbFromHex(SEED)));
  tokens.warning = hexFromArgb(warning.tone(dark ? 80 : 40));
  return tokens;
}

const block = (selector, tokens, indent = "") =>
  [`${indent}${selector} {`, ...Object.entries(tokens).map(([k, v]) => `${indent}  --${k}: ${v};`), `${indent}}`].join("\n");

export const START = "/* theme:start (generated by scripts/generate-theme.mjs; edit the script, not these lines) */";
export const END = "/* theme:end */";

/** The generated part of globals.css: light on :root, dark under the phone's setting, and `.paper` always light. */
export function themeCss() {
  const light = themeTokens(false);
  const dark = themeTokens(true);
  return [
    START,
    block(":root", light),
    "",
    "@media (prefers-color-scheme: dark) {",
    block(":root", dark, "  "),
    "}",
    "",
    "/* Anything that stands for paper (the printed bulletin) stays light in dark mode. */",
    block(".paper", light),
    END,
  ].join("\n");
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const css = themeCss();
  if (process.argv.includes("--write")) {
    const path = fileURLToPath(new URL("../src/app/globals.css", import.meta.url));
    const text = readFileSync(path, "utf8");
    const from = text.indexOf("/* theme:start");
    const to = text.indexOf(END);
    if (from < 0 || to < from) throw new Error("globals.css has no theme:start/theme:end block");
    writeFileSync(path, text.slice(0, from) + css + text.slice(to + END.length));
    console.log("globals.css: theme block written");
  } else {
    console.log(css);
  }
}

// The app icon (Theme PR 1): a white cross on the theme's teal. Writes the
// SVG favicon, favicon.ico, the Apple touch icon and the web manifest's
// icons, all from the geometry below, with sharp (installed with Next.js;
// no network). Run `node scripts/generate-icons.mjs` after changing it and
// commit the files it writes.
import { writeFileSync } from "node:fs";
import { fileURLToPath } from "node:url";

import sharp from "sharp";

const TEAL = "#006A60"; // the seed, the light primary
const CROSS = '<path fill="#FFFFFF" d="M226 112h60v68h76v60h-76v160h-60V240h-76v-60h76z"/>';

/** `rounded`: the favicon and the manifest's "any" icons; full bleed for the Apple and maskable icons (the system rounds them). */
export function iconSvg(rounded) {
  const rx = rounded ? ' rx="112"' : "";
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 512 512"><rect width="512" height="512"${rx} fill="${TEAL}"/>${CROSS}</svg>\n`;
}

const at = (path) => fileURLToPath(new URL(`../${path}`, import.meta.url));
const png = (svg, size) => sharp(Buffer.from(svg)).resize(size, size).png().toBuffer();

/** An .ico holding PNG images (every current browser reads these). */
function ico(images) {
  const header = Buffer.alloc(6);
  header.writeUInt16LE(0, 0);
  header.writeUInt16LE(1, 2);
  header.writeUInt16LE(images.length, 4);
  let offset = 6 + 16 * images.length;
  const entries = images.map(({ size, data }) => {
    const e = Buffer.alloc(16);
    e.writeUInt8(size % 256, 0);
    e.writeUInt8(size % 256, 1);
    e.writeUInt16LE(1, 4);
    e.writeUInt16LE(32, 6);
    e.writeUInt32LE(data.length, 8);
    e.writeUInt32LE(offset, 12);
    offset += data.length;
    return e;
  });
  return Buffer.concat([header, ...entries, ...images.map((i) => i.data)]);
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const rounded = iconSvg(true);
  const full = iconSvg(false);
  writeFileSync(at("src/app/icon.svg"), rounded);
  writeFileSync(at("src/app/favicon.ico"), ico(await Promise.all([16, 32, 48].map(async (size) => ({ size, data: await png(rounded, size) })))));
  writeFileSync(at("src/app/apple-icon.png"), await png(full, 180));
  writeFileSync(at("public/icons/icon-192.png"), await png(rounded, 192));
  writeFileSync(at("public/icons/icon-512.png"), await png(rounded, 512));
  writeFileSync(at("public/icons/icon-maskable-512.png"), await png(full, 512));
  console.log("icons written: icon.svg, favicon.ico, apple-icon.png, icons/icon-192.png, icons/icon-512.png, icons/icon-maskable-512.png");
}
